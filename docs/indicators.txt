f_blockBar(float score) =>
    int    n = math.min(math.max(math.round(score), 0), 10)
    string s = ''
    for i = 1 to 10
        s += i <= n ? '█' : '░'
    s + ' ' + str.tostring(math.round(score / 10.0 * 100.0), '##0') + '%'

// ── HELPER FUNCTIONS ─────────────────────────────────────────────────────────
f_pivotHigh(series float src, simple int len) =>
    float candidate = src[len]
    bool  isValid   = true
    for i = 1 to len * 2
        if i != len and src[i] >= candidate
            isValid := false
    isValid ? candidate : na

f_pivotLow(series float src, simple int len) =>
    float candidate = src[len]
    bool  isValid   = true
    for i = 1 to len * 2
        if i != len and src[i] <= candidate
            isValid := false
    isValid ? candidate : na

f_project(float x1, float y1, float x2, float y2, float target_x) =>
    x2 == x1 ? y1 : y1 + ((y2 - y1) / (x2 - x1)) * (target_x - x1)

f_isNear(float v1, float v2, float tol_pct) =>
    math.abs(v1 - v2) <= ((v1 + v2) / 2) * tol_pct

f_neck_at_break(int nl_i, float nl_p, int nr_i, float nr_p, int b_idx) =>
    na(nl_i) and na(nr_i) ? na :
     na(nl_i) ? nr_p :
     na(nr_i) ? nl_p :
     nl_i == nr_i ? nl_p : f_project(nl_i, nl_p, nr_i, nr_p, b_idx)

f_isValidSize(float height, float price) =>
    height > math.max(price * min_size_pct, atr_val * min_atr_mult)

f_apply_buffer(float price, bool isBullish) =>
    isBullish ? price - (atr_val * stop_buffer) : price + (atr_val * stop_buffer)

f_sl_pat(float anchor, bool isBull, float entry_p, float target_p) =>
    if use_sl_pct_tgt and not na(entry_p) and not na(target_p)
        float _dist = math.abs(target_p - entry_p)
        isBull ? entry_p - _dist * sl_pct_tgt : entry_p + _dist * sl_pct_tgt
    else
        f_apply_buffer(anchor, isBull)

f_get_entry_price(int b_idx) =>
    int offs = bar_index - b_idx
    offs >= 1 ? open[offs] : na

f_get_break_idx_precise(float x1, float y1, float x2, float y2, bool isUp) =>
    int   b_idx      = na
    int   scan_limit = math.max(min_brk_bar, lb_right) + 10
    float m          = x2 == x1 ? 0.0 : (y2 - y1) / (x2 - x1)
    for i = scan_limit to 1
        if (bar_index - i) < 0
            continue
        float proj_price  = x2 == x1 ? y1 : y1 + m * ((bar_index - i) - x1)
        float margin      = use_atr_brk ? atr_val[i] * break_atr : proj_price * break_pct
        float up_thresh   = proj_price + margin
        float down_thresh = proj_price - margin
        float check_val   = req_close_brk ? close[i] : isUp ? high[i] : low[i]
        bool  is_broken   = isUp ? check_val > up_thresh : check_val < down_thresh
        if req_body_brk and is_broken and not show_ignore_conf
            is_broken := isUp ? open[i] > up_thresh and close[i] > up_thresh : open[i] < down_thresh and close[i] < down_thresh
        if is_broken
            bool confirmed = true
            if brk_confirm_bars > 1 and not show_ignore_conf
                for k = 1 to brk_confirm_bars - 1
                    int check_idx = i - k
                    if check_idx < 0
                        confirmed := false
                        break
                    float proj_k   = x2 == x1 ? y1 : y1 + m * ((bar_index - check_idx) - x1)
                    float margin_k = use_atr_brk ? atr_val[check_idx] * break_atr : proj_k * break_pct
                    if isUp and close[check_idx] <= proj_k + margin_k
                        confirmed := false
                        break
                    else if not isUp and close[check_idx] >= proj_k - margin_k
                        confirmed := false
                        break
            if confirmed
                b_idx := bar_index - i
                break
    b_idx

f_validate_pattern(int b_idx, float entry_p, float stop_p, float target_p) =>
    if show_ignore_conf
        true
    else
        bool valid = true
        if req_vol and not na(b_idx)
            int offs = bar_index - b_idx
            if offs >= 0 and offs < bar_index
                float sma_at_offs = vol_sma[offs]
                if not na(sma_at_offs) and volume[offs] <= sma_at_offs * vol_mult
                    valid := false
        if valid and not na(entry_p)
            float total_cost = SLIP_COST_ABS + entry_p * (commission_pct / 100) * 2
            float risk       = math.abs(entry_p - stop_p)
            float reward_adj = math.max(0, math.abs(target_p - entry_p) - total_cost)
            if risk == 0 or reward_adj / risk < min_rr
                valid := false
        valid

f_fill_sliced(int ux1, float uy1, int ux2, float uy2, int lx1, float ly1, int lx2, float ly2, color fill_col, array<line> lns_arr) =>
    line ln_top = line.new(ux1, uy1, ux2, uy2, color=color.new(fill_col, 100), width=1)
    line ln_bot = line.new(lx1, ly1, lx2, ly2, color=color.new(fill_col, 100), width=1)
    linefill.new(ln_top, ln_bot, color.new(fill_col, fill_transp))
    array.push(lns_arr, ln_top)
    array.push(lns_arr, ln_bot)

f_manage_drawings(int max_items) =>
    while array.size(pat_history) > max_items
        PatternDrawings old_pat = array.shift(pat_history)
        for _ln in old_pat.lns
            if not na(_ln)
                line.delete(_ln)
        for _lb in old_pat.lbls
            if not na(_lb)
                label.delete(_lb)

// ── PATTERN DETECTION FUNCTIONS ──────────────────────────────────────────────
detectHS() =>
    res = EMPTY_RES
    _has_hs  = pivotHighs.size() >= 3 and pivotLows.size() >= 2
    _has_ihs = pivotLows.size() >= 3 and pivotHighs.size() >= 2
    _rs       = _has_hs  ? pivotHighs.get(0) : Pivot.new(na, na)
    _head     = _has_hs  ? pivotHighs.get(1) : Pivot.new(na, na)
    _ls       = _has_hs  ? pivotHighs.get(2) : Pivot.new(na, na)
    _l_neck_r = _has_hs  ? pivotLows.get(0)  : Pivot.new(na, na)
    _l_neck_l = _has_hs  ? pivotLows.get(1)  : Pivot.new(na, na)
    _rs_inv   = _has_ihs ? pivotLows.get(0)   : Pivot.new(na, na)
    _head_inv = _has_ihs ? pivotLows.get(1)   : Pivot.new(na, na)
    _ls_inv   = _has_ihs ? pivotLows.get(2)   : Pivot.new(na, na)
    _h_neck_r = _has_ihs ? pivotHighs.get(0)  : Pivot.new(na, na)
    _h_neck_l = _has_ihs ? pivotHighs.get(1)  : Pivot.new(na, na)
    int   _b_hs      = f_get_break_idx_precise(_l_neck_l.i, _l_neck_l.p, _l_neck_r.i, _l_neck_r.p, false)
    int   _b_ihs     = f_get_break_idx_precise(_h_neck_l.i, _h_neck_l.p, _h_neck_r.i, _h_neck_r.p, true)
    float _ep_hs     = f_get_entry_price(_b_hs)
    float _na_hs     = f_neck_at_break(_l_neck_l.i, _l_neck_l.p, _l_neck_r.i, _l_neck_r.p, nz(_b_hs))
    float _neck_avg  = (nz(_l_neck_r.p) + nz(_l_neck_l.p)) / 2
    float _ht_hs     = nz(_head.p) - _neck_avg
    float _tp_hs     = _na_hs - _ht_hs
    float _sp_hs     = f_sl_pat(_rs.p, false, _ep_hs, _tp_hs)
    bool  _val_hs    = f_validate_pattern(_b_hs, _ep_hs, _sp_hs, _tp_hs)
    float _ep_ihs    = f_get_entry_price(_b_ihs)
    float _na_ihs    = f_neck_at_break(_h_neck_l.i, _h_neck_l.p, _h_neck_r.i, _h_neck_r.p, nz(_b_ihs))
    float _neck_avg_inv = (nz(_h_neck_r.p) + nz(_h_neck_l.p)) / 2
    float _ht_ihs    = _neck_avg_inv - nz(_head_inv.p)
    float _tp_ihs    = _na_ihs + _ht_ihs
    float _sp_ihs    = f_sl_pat(_rs_inv.p, true, _ep_ihs, _tp_ihs)
    bool  _val_ihs   = f_validate_pattern(_b_ihs, _ep_ihs, _sp_ihs, _tp_ihs)
    if _has_hs
        if _rs.i > _l_neck_r.i and _l_neck_r.i > _head.i and _head.i > _l_neck_l.i and _l_neck_l.i > _ls.i
            if _head.p > _rs.p and _head.p > _ls.p and f_isNear(_ls.p, _rs.p, sym_tol)
                if f_isValidSize(_ht_hs, close)
                    if not na(_b_hs)
                        if not na(_ep_hs) and _val_hs and show_hs_std
                            res         := PatternResult.new(true, "Head & Shoulders", false, _ep_hs, _sp_hs, _tp_hs, na, na, _ls.i, _b_hs, _ls.i, _head.p, _b_hs, _na_hs, _l_neck_l.i, _l_neck_l.p, _b_hs, _na_hs, na, na, na, na, _l_neck_l.i, _l_neck_l.p, _b_hs, _na_hs)
    if not res.detected and _has_ihs
        if _rs_inv.i > _h_neck_r.i and _h_neck_r.i > _head_inv.i and _head_inv.i > _h_neck_l.i and _h_neck_l.i > _ls_inv.i
            if _head_inv.p < _rs_inv.p and _head_inv.p < _ls_inv.p and f_isNear(_ls_inv.p, _rs_inv.p, sym_tol)
                if f_isValidSize(_ht_ihs, close)
                    if not na(_b_ihs)
                        if not na(_ep_ihs) and _val_ihs and show_hs_inv
                            res         := PatternResult.new(true, "Inv Head & Shoulders", true, _ep_ihs, _sp_ihs, _tp_ihs, na, na, _ls_inv.i, _b_ihs, _h_neck_l.i, _h_neck_l.p, _b_ihs, _na_ihs, _ls_inv.i, _head_inv.p, _b_ihs, _na_ihs, _h_neck_l.i, _h_neck_l.p, _b_ihs, _na_ihs, na, na, na, na)
    res

detectDouble() =>
    res = EMPTY_RES
    _has_dt  = pivotHighs.size() >= 2 and pivotLows.size() >= 1
    _has_db  = pivotLows.size() >= 2 and pivotHighs.size() >= 1
    _p1      = _has_dt ? pivotHighs.get(0) : Pivot.new(na, na)
    _p2      = _has_dt ? pivotHighs.get(1) : Pivot.new(na, na)
    _mid     = _has_dt ? pivotLows.get(0)  : Pivot.new(na, na)
    _p1b     = _has_db ? pivotLows.get(0)   : Pivot.new(na, na)
    _p2b     = _has_db ? pivotLows.get(1)   : Pivot.new(na, na)
    _mid_b   = _has_db ? pivotHighs.get(0)  : Pivot.new(na, na)
    int   _b_dt   = f_get_break_idx_precise(_mid.i, _mid.p, bar_index, _mid.p, false)
    int   _b_db   = f_get_break_idx_precise(_mid_b.i, _mid_b.p, bar_index, _mid_b.p, true)
    float _ep_dt  = f_get_entry_price(_b_dt)
    float _avg_top = (nz(_p1.p) + nz(_p2.p)) / 2
    float _ht_dt  = _avg_top - nz(_mid.p)
    float _tp_dt  = nz(_mid.p) - _ht_dt
    float _sp_dt  = f_sl_pat(math.max(nz(_p1.p), nz(_p2.p)), false, _ep_dt, _tp_dt)
    bool  _val_dt = f_validate_pattern(_b_dt, _ep_dt, _sp_dt, _tp_dt)
    float _ep_db  = f_get_entry_price(_b_db)
    float _avg_bot = (nz(_p1b.p) + nz(_p2b.p)) / 2
    float _ht_db  = nz(_mid_b.p) - _avg_bot
    float _tp_db  = nz(_mid_b.p) + _ht_db
    float _sp_db  = f_sl_pat(math.min(nz(_p1b.p), nz(_p2b.p)), true, _ep_db, _tp_db)
    bool  _val_db = f_validate_pattern(_b_db, _ep_db, _sp_db, _tp_db)
    if _has_dt
        if _p1.i > _mid.i and _mid.i > _p2.i and f_isNear(_p1.p, _p2.p, lvl_tol)
            if f_isValidSize(_ht_dt, close)
                if not na(_b_dt)
                    if not na(_ep_dt) and _val_dt and show_dt_top

    if not res.detected and _has_db
        if _p1b.i > _mid_b.i and _mid_b.i > _p2b.i and f_isNear(_p1b.p, _p2b.p, lvl_tol)

detectTriple() =>
    res = EMPTY_RES
    _has_tt  = pivotHighs.size() >= 3 and pivotLows.size() >= 2
    _has_tb  = pivotLows.size() >= 3 and pivotHighs.size() >= 2
    _h1      = _has_tt ? pivotHighs.get(0) : Pivot.new(na, na)
    _h2      = _has_tt ? pivotHighs.get(1) : Pivot.new(na, na)
    _h3      = _has_tt ? pivotHighs.get(2) : Pivot.new(na, na)
    _l1t     = _has_tt ? pivotLows.get(0)  : Pivot.new(na, na)
    _l2t     = _has_tt ? pivotLows.get(1)  : Pivot.new(na, na)
    _l1b     = _has_tb ? pivotLows.get(0)   : Pivot.new(na, na)
    _l2b     = _has_tb ? pivotLows.get(1)   : Pivot.new(na, na)
    _l3b     = _has_tb ? pivotLows.get(2)   : Pivot.new(na, na)
    _h1b     = _has_tb ? pivotHighs.get(0)  : Pivot.new(na, na)
    _h2b     = _has_tb ? pivotHighs.get(1)  : Pivot.new(na, na)
    float _neck_tt    = math.min(nz(_l1t.p), nz(_l2t.p))
    float _ht_tt      = math.max(nz(_h1.p), math.max(nz(_h2.p), nz(_h3.p))) - _neck_tt
    float _neck_tb    = math.max(nz(_h1b.p), nz(_h2b.p))
    float _ht_tb      = _neck_tb - math.min(nz(_l1b.p), math.min(nz(_l2b.p), nz(_l3b.p)))
    int   _b_tt       = f_get_break_idx_precise(_l2t.i, _l2t.p, _l1t.i, _l1t.p, false)
    int   _b_tb       = f_get_break_idx_precise(_h2b.i, _neck_tb, _h1b.i, _neck_tb, true)
    float _na_tt      = f_neck_at_break(_l2t.i, _l2t.p, _l1t.i, _l1t.p, nz(_b_tt))
    float _na_tb      = f_neck_at_break(_h2b.i, _neck_tb, _h1b.i, _neck_tb, nz(_b_tb))
    float _ep_tt      = f_get_entry_price(_b_tt)
    float _ep_tb      = f_get_entry_price(_b_tb)
    float _tp_tt      = _na_tt - _ht_tt
    float _tp_tb      = _na_tb + _ht_tb
    float _sp_tt      = f_sl_pat(math.max(nz(_h1.p), math.max(nz(_h2.p), nz(_h3.p))), false, _ep_tt, _tp_tt)
    float _sp_tb      = f_sl_pat(math.min(nz(_l1b.p), math.min(nz(_l2b.p), nz(_l3b.p))), true, _ep_tb, _tp_tb)
    bool  _val_tt     = f_validate_pattern(_b_tt, _ep_tt, _sp_tt, _tp_tt)
    bool  _val_tb     = f_validate_pattern(_b_tb, _ep_tb, _sp_tb, _tp_tb)
    if _has_tt
        if _h1.i > _l1t.i and _l1t.i > _h2.i and _h2.i > _l2t.i and _l2t.i > _h3.i
            if f_isNear(_h1.p, _h2.p, lvl_tol) and f_isNear(_h2.p, _h3.p, lvl_tol)
                if f_isValidSize(_ht_tt, close)
                    if not na(_b_tt)
                        if not na(_ep_tt) and _val_tt and show_tt_top
                            float _avg_top_tt = math.max(_h1.p, math.max(_h2.p, _h3.p))
                            res := PatternResult.new(true, "Triple Top", false, _ep_tt, _sp_tt, _tp_tt, na, na, _h3.i, _b_tt, _h3.i, _avg_top_tt, _b_tt, _avg_top_tt, _l2t.i, _neck_tt, _b_tt, _na_tt, na, na, na, na, _l2t.i, _neck_tt, _b_tt, _na_tt)
    if not res.detected and _has_tb
        if _l1b.i > _h1b.i and _h1b.i > _l2b.i and _l2b.i > _h2b.i and _h2b.i > _l3b.i
            if f_isNear(_l1b.p, _l2b.p, lvl_tol) and f_isNear(_l2b.p, _l3b.p, lvl_tol)
                if f_isValidSize(_ht_tb, close)
                    if not na(_b_tb)
                        if not na(_ep_tb) and _val_tb and show_tt_bot
                            float _avg_bot_tt = math.min(_l1b.p, math.min(_l2b.p, _l3b.p))
                            res := PatternResult.new(true, "Triple Bottom", true, _ep_tb, _sp_tb, _tp_tb, na, na, _l3b.i, _b_tb, _h2b.i, _neck_tb, _b_tb, _na_tb, _l3b.i, _avg_bot_tt, _b_tb, _avg_bot_tt, _h2b.i, _neck_tb, _b_tb, _na_tb, na, na, na, na)
    res

    float _ep_inv     = f_get_entry_price(_b_inv)

detectFlagPennant() =>
    res = EMPTY_RES
    _has = pivotHighs.size() >= 2 and pivotLows.size() >= 2
    _h1  = _has ? pivotHighs.get(0) : Pivot.new(na, na)
    _h2  = _has ? pivotHighs.get(1) : Pivot.new(na, na)
    _l1  = _has ? pivotLows.get(0)  : Pivot.new(na, na)
    _l2  = _has ? pivotLows.get(1)  : Pivot.new(na, na)
    int   _start_bar  = math.min(nz(_h2.i), nz(_l2.i))
    int   _pole_len   = math.min(200, bar_index - _start_bar + 1)
    float _pole_move  = ta.highest(high, _pole_len) - ta.lowest(low, _pole_len)
    float _slope_u    = _h1.i == _h2.i ? 0.0 : (nz(_h1.p) - nz(_h2.p)) / math.max(1, _h1.i - _h2.i)
    float _slope_l    = _l1.i == _l2.i ? 0.0 : (nz(_l1.p) - nz(_l2.p)) / math.max(1, _l1.i - _l2.i)
    bool  _parallel   = f_isNear(_slope_u, _slope_l, 0.2)
    int   _b_bull     = f_get_break_idx_precise(_h2.i, _h2.p, _h1.i, _h1.p, true)
    int   _b_bear     = f_get_break_idx_precise(_l2.i, _l2.p, _l1.i, _l1.p, false)
    float _ub_bull    = f_project(_h2.i, _h2.p, _h1.i, _h1.p, nz(_b_bull))
    float _lb_bull    = f_project(_l2.i, _l2.p, _l1.i, _l1.p, nz(_b_bull))
    float _ub_bear    = f_project(_h2.i, _h2.p, _h1.i, _h1.p, nz(_b_bear))
    float _lb_bear    = f_project(_l2.i, _l2.p, _l1.i, _l1.p, nz(_b_bear))
    float _ep_bull    = f_get_entry_price(_b_bull)
    float _ep_bear    = f_get_entry_price(_b_bear)
    float _tp_bull    = _ub_bull + _pole_move
    float _tp_bear    = _lb_bear - _pole_move
    float _sp_bull    = f_sl_pat(_lb_bull, true, _ep_bull, _tp_bull)
    float _sp_bear    = f_sl_pat(_ub_bear, false, _ep_bear, _tp_bear)
    bool  _val_bull   = f_validate_pattern(_b_bull, _ep_bull, _sp_bull, _tp_bull)
    bool  _val_bear   = f_validate_pattern(_b_bear, _ep_bear, _sp_bear, _tp_bear)
    if _has and _pole_move > atr_val * 3
        if _slope_u < 0 and _slope_l < 0
            if not na(_b_bull)
                if not na(_ep_bull) and _val_bull
                    if (_parallel and show_flag_bull) or (not _parallel and show_pennant_bull)
                        res := PatternResult.new(true, _parallel ? "Bullish Flag" : "Bullish Pennant", true, _ep_bull, _sp_bull, _tp_bull, na, na, _h2.i, _b_bull, _h2.i, _h2.p, _b_bull, _ub_bull, _l2.i, _l2.p, _b_bull, _lb_bull, na, na, na, na, na, na, na, na)
        else if _slope_u > 0 and _slope_l > 0
            if not na(_b_bear)
                if not na(_ep_bear) and _val_bear
                    if (_parallel and show_flag_bear) or (not _parallel and show_pennant_bear)
                        res := PatternResult.new(true, _parallel ? "Bearish Flag" : "Bearish Pennant", false, _ep_bear, _sp_bear, _tp_bear, na, na, _h2.i, _b_bear, _h2.i, _h2.p, _b_bear, _ub_bear, _l2.i, _l2.p, _b_bear, _lb_bear, na, na, na, na, na, na, na, na)
    if res.detected
        int   _sb  = math.min(_h2.i, _l2.i)
        int   _bi  = res.breakout_idx
        float _ubs = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _bi)
        float _lbs = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _bi)
        float _uss = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _sb)
        float _lss = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _sb)
        res.fux1  := _sb
        res.fuy1  := _uss
        res.fux2  := _bi
        res.fuy2  := _ubs
        res.flx1  := _sb
        res.fly1  := _lss
        res.flx2  := _bi
        res.fly2  := _lbs
        res.lux1  := _sb
        res.luy1  := _uss
        res.lux2  := _bi
        res.luy2  := _ubs
        res.llx1  := _sb
        res.lly1  := _lss
        res.llx2  := _bi
        res.lly2  := _lbs
    res

detectWedge() =>

    _h2  = _has ? pivotHighs.get(1) : Pivot.new(na, na)
    _l1  = _has ? pivotLows.get(0)  : Pivot.new(na, na)
    _l2  = _has ? pivotLows.get(1)  : Pivot.new(na, na)

    float _slope_l         = _l1.i == _l2.i ? 0.0 : (nz(_l1.p) - nz(_l2.p)) / math.max(1, _l1.i - _l2.i)
    int   _start_idx_wedge = math.min(nz(_h2.i), nz(_l2.i))
    float _wedge_base_h    = math.abs(f_project(_h2.i, _h2.p, _h1.i, _h1.p, _start_idx_wedge) - f_project(_l2.i, _l2.p, _l1.i, _l1.p, _start_idx_wedge))
    float _proj_u_now      = f_project(_h2.i, _h2.p, _h1.i, _h1.p, bar_index)
    float _proj_l_now      = f_project(_l2.i, _l2.p, _l1.i, _l1.p, bar_index)
    float _height_now      = _proj_u_now - _proj_l_now
    int   _b_fall          = f_get_break_idx_precise(_h2.i, _h2.p, _h1.i, _h1.p, true)

    float _ep_rise         = f_get_entry_price(_b_rise)

    float _tp_rise         = nz(_l2.p)

    float _sp_rise         = f_sl_pat(_ub_rise, false, _ep_rise, _tp_rise)

    bool  _val_rise        = f_validate_pattern(_b_rise, _ep_rise, _sp_rise, _tp_rise)
    if _has and _height_now > 0
        if _slope_u < 0 and _slope_l < 0 and _slope_l < _slope_u
            if not na(_b_fall)
                if not na(_ep_fall) and _val_fall and show_wedge_fall
                    res := PatternResult.new(true, "Falling Wedge", true, _ep_fall, _sp_fall, _tp_fall, na, na, _h2.i, _b_fall)
        else if _slope_u > 0 and _slope_l > 0 and _slope_l > _slope_u
            if not na(_b_rise)
                if not na(_ep_rise) and _val_rise and show_wedge_rise

    if res.detected
        int   _sb  = math.min(_h2.i, _l2.i)
        int   _bi  = res.breakout_idx
        float _ubs = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _bi)
        float _lbs = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _bi)
        float _uss = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _sb)
        float _lss = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _sb)
        res.fux1  := _h2.i
        res.fuy1  := _h2.p
        res.fux2  := _bi
        res.fuy2  := _ubs
        res.flx1  := _l2.i
        res.fly1  := _l2.p
        res.flx2  := _bi
        res.fly2  := _lbs
        res.lux1  := _sb
        res.luy1  := _uss
        res.lux2  := _bi
        res.luy2  := _ubs
        res.llx1  := _sb
        res.lly1  := _lss

detectTriangles() =>
    res = EMPTY_RES
    _has = pivotHighs.size() >= 2 and pivotLows.size() >= 2
    _h1  = _has ? pivotHighs.get(0) : Pivot.new(na, na)
    _h2  = _has ? pivotHighs.get(1) : Pivot.new(na, na)
    _l1  = _has ? pivotLows.get(0)  : Pivot.new(na, na)
    _l2  = _has ? pivotLows.get(1)  : Pivot.new(na, na)
    float _slope_u      = _h1.i == _h2.i ? 0.0 : (nz(_h1.p) - nz(_h2.p)) / math.max(1, _h1.i - _h2.i)
    float _slope_l      = _l1.i == _l2.i ? 0.0 : (nz(_l1.p) - nz(_l2.p)) / math.max(1, _l1.i - _l2.i)
    int   _start_tri    = math.min(nz(_h2.i), nz(_l2.i))
    float _base_height  = math.abs(f_project(_h2.i, _h2.p, _h1.i, _h1.p, _start_tri) - f_project(_l2.i, _l2.p, _l1.i, _l1.p, _start_tri))
    float _flat_tol     = close * 0.0005
    bool  _converging   = f_project(_h2.i, _h2.p, _h1.i, _h1.p, bar_index) > f_project(_l2.i, _l2.p, _l1.i, _l1.p, bar_index)
    int   _b_bull       = f_get_break_idx_precise(_h2.i, _h2.p, _h1.i, _h1.p, true)
    int   _b_bear       = f_get_break_idx_precise(_l2.i, _l2.p, _l1.i, _l1.p, false)
    float _ub_bull      = f_project(_h2.i, _h2.p, _h1.i, _h1.p, nz(_b_bull))
    float _lb_bull      = f_project(_l2.i, _l2.p, _l1.i, _l1.p, nz(_b_bull))
    float _ub_bear      = f_project(_h2.i, _h2.p, _h1.i, _h1.p, nz(_b_bear))
    float _lb_bear      = f_project(_l2.i, _l2.p, _l1.i, _l1.p, nz(_b_bear))
    float _ep_bull      = f_get_entry_price(_b_bull)
    float _ep_bear      = f_get_entry_price(_b_bear)
    float _tp_bull      = _ub_bull + _base_height
    float _tp_bear      = _lb_bear - _base_height
    float _sp_bull      = f_sl_pat(_lb_bull, true, _ep_bull, _tp_bull)
    float _sp_bear      = f_sl_pat(_ub_bear, false, _ep_bear, _tp_bear)
    bool  _val_bull     = f_validate_pattern(_b_bull, _ep_bull, _sp_bull, _tp_bull)
    bool  _val_bear     = f_validate_pattern(_b_bear, _ep_bear, _sp_bear, _tp_bear)
    if _has and _converging and f_isValidSize(_base_height, close)
        if (_slope_u < 0 and _slope_l > 0) or (math.abs(_slope_u) < _flat_tol and _slope_l > 0)
            if not na(_b_bull)
                if not na(_ep_bull) and _val_bull and show_tri_gen
                    bool _isBullish = math.abs(_slope_u) < _flat_tol
                    res := PatternResult.new(true, _isBullish ? "Ascending Triangle" : "Symmetrical Triangle", true, _ep_bull, _sp_bull, _tp_bull, na, na, _h2.i, _b_bull)
        if not res.detected and ((_slope_u < 0 and _slope_l > 0) or (_slope_u < 0 and math.abs(_slope_l) < _flat_tol))
            if not na(_b_bear)
                if not na(_ep_bear) and _val_bear and show_tri_gen
                    bool _isBearish = math.abs(_slope_l) < _flat_tol
                    res := PatternResult.new(true, _isBearish ? "Descending Triangle" : "Symmetrical Triangle", false, _ep_bear, _sp_bear, _tp_bear, na, na, _h2.i, _b_bear)
    if res.detected
        int   _sb  = math.min(_h2.i, _l2.i)
        int   _bi  = res.breakout_idx
        float _ubs = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _bi)
        float _lbs = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _bi)
        float _uss = f_project(_h2.i, _h2.p, _h1.i, _h1.p, _sb)
        float _lss = f_project(_l2.i, _l2.p, _l1.i, _l1.p, _sb)
        res.fux1  := _h2.i
        res.fuy1  := _h2.p
        res.fux2  := _bi
        res.fuy2  := _ubs
        res.flx1  := _l2.i
        res.fly1  := _l2.p
        res.flx2  := _bi
        res.fly2  := _lbs
        res.lux1  := _sb
        res.luy1  := _uss
        res.lux2  := _bi
        res.luy2  := _ubs
        res.llx1  := _sb
        res.lly1  := _lss
        res.llx2  := _bi
        res.lly2  := _lbs
    res

detectRectangles() =>
    res = EMPTY_RES
    _has = pivotHighs.size() >= 2 and pivotLows.size() >= 2
    _h1  = _has ? pivotHighs.get(0) : Pivot.new(na, na)
    _h2  = _has ? pivotHighs.get(1) : Pivot.new(na, na)
    _l1  = _has ? pivotLows.get(0)  : Pivot.new(na, na)
    _l2  = _has ? pivotLows.get(1)  : Pivot.new(na, na)
    float _slope_u      = _h1.i == _h2.i ? 0.0 : (nz(_h1.p) - nz(_h2.p)) / math.max(1, _h1.i - _h2.i)
    float _slope_l      = _l1.i == _l2.i ? 0.0 : (nz(_l1.p) - nz(_l2.p)) / math.max(1, _l1.i - _l2.i)
    float _flat_tol_r   = atr_val * 0.05
    bool  _is_flat      = math.abs(_slope_u) < _flat_tol_r and math.abs(_slope_l) < _flat_tol_r
    int   _start_rect   = math.min(nz(_h2.i), nz(_l2.i))
    float _avg_top      = (nz(_h1.p) + nz(_h2.p)) / 2
    float _avg_bot      = (nz(_l1.p) + nz(_l2.p)) / 2
    float _height       = _avg_top - _avg_bot
    int   _b_bull       = f_get_break_idx_precise(_start_rect, _avg_top, bar_index, _avg_top, true)
    int   _b_bear       = f_get_break_idx_precise(_start_rect, _avg_bot, bar_index, _avg_bot, false)
    float _ep_bull      = f_get_entry_price(_b_bull)
    float _ep_bear      = f_get_entry_price(_b_bear)
    float _tp_bull      = _avg_top + _height
    float _tp_bear      = _avg_bot - _height
    float _sp_bull      = f_sl_pat(_avg_bot, true, _ep_bull, _tp_bull)
    float _sp_bear      = f_sl_pat(_avg_top, false, _ep_bear, _tp_bear)
    bool  _val_bull     = f_validate_pattern(_b_bull, _ep_bull, _sp_bull, _tp_bull)
    bool  _val_bear     = f_validate_pattern(_b_bear, _ep_bear, _sp_bear, _tp_bear)
    if _has and _is_flat and f_isNear(_slope_u, _slope_l, 0.5) and f_isValidSize(_height, close)
        if not na(_b_bull)
            if not na(_ep_bull) and _val_bull and show_rect_gen
                res := PatternResult.new(true, "Rectangle", true, _ep_bull, _sp_bull, _tp_bull, na, na, _h2.i, _b_bull, _start_rect, _avg_top, _b_bull, _avg_top, _start_rect, _avg_bot, _b_bull, _avg_bot, na, na, na, na, na, na, na, na)
        if not res.detected and not na(_b_bear)
            if not na(_ep_bear) and _val_bear and show_rect_gen
                res := PatternResult.new(true, "Rectangle", false, _ep_bear, _sp_bear, _tp_bear, na, na, _h2.i, _b_bear, _start_rect, _avg_top, _b_bear, _avg_top, _start_rect, _avg_bot, _b_bear, _avg_bot, na, na, na, na, na, na, na, na)
    if res.detected
        int   _sb  = math.min(_h2.i, _l2.i)
        int   _bi  = res.breakout_idx
        res.lux1  := _sb
        res.luy1  := _avg_top
        res.lux2  := _bi
        res.luy2  := _avg_top
        res.llx1  := _sb
        res.lly1  := _avg_bot
        res.llx2  := _bi
        res.lly2  := _avg_bot
    res

method getZigzagAndPattern(Scanner this, int length, int depth, array<o.OHLC> ohlcArray, int offset=0)=>
    var zg.Zigzag zigzag = zg.Zigzag.new(length, depth, 0)
    var map<int, int> lastDBar = map.new<int, int>()
    zigzag.calculate(array.from(highSource, lowSource))

    var validPatterns = 0
    mlzigzag = zigzag
    if(zigzag.flags.newPivot)
        while(mlzigzag.zigzagPivots.size() >= 6+offset)
            lastBar = mlzigzag.zigzagPivots.first().point.index
            lastDir = int(math.sign(mlzigzag.zigzagPivots.first().dir))
            if(lastDBar.contains(mlzigzag.level)? lastDBar.get(mlzigzag.level) < lastBar : true)
                lastDBar.put(mlzigzag.level, lastBar)
                [valid, currentPattern] = mlzigzag.find(this.sProperties, this.dProperties, this.patterns, ohlcArray)
                if(valid)
                    validPatterns+=1
                    currentPattern.draw()
                    this.patterns.push(currentPattern, maxPatterns)
                    alert('New Pattern Alert')
            else
                break
            mlzigzag := mlzigzag.nextlevel()
    true

        this.getZigzagAndPattern(zigzagLength3, depth3, ohlcArray)
    if(useZigzag4)
        this.getZigzagAndPattern(zigzagLength4, depth4, ohlcArray)
