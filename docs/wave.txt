// â”€â”€â”€ Projection Function â”€â”€â”€
drawProjection(startP, startT, w1Len, originP, simI, isBull, col, style, currentPrice) =>
    len1 = w1Len
    len2 = len1 * 0.618
    len3 = len1 * 1.618
    len4 = len3 * 0.382
    len5 = len1

    dir0 = isBull ? 1.0 : -1.0
    r_w1 = dir0 * len1
    r_w2 = r_w1 - dir0 * len2
    r_w3 = r_w2 + dir0 * len3
    r_w4 = r_w3 - dir0 * len4
    r_w5 = r_w4 + dir0 * len5
    w5   = originP + r_w5
    cycle0Net = w5 - originP

    simP = startP
    simT = startT
    barSp = 22

    // Flag to determine when to start drawing
    bool isFirstForwardLine = true

    for s = 1 to 12
        effAbsIdx = simI + s
        cycleNum  = int(math.floor(effAbsIdx / 9))
        currIdx   = effAbsIdx % 9

        cycleSign = (cycleNum % 2 == 0) ? 1.0 : -1.0
        activeDir = (isBull ? 1.0 : -1.0) * cycleSign

        float priorCyclesMove = 0.0
        if cycleNum == 1
            priorCyclesMove := cycle0Net
        else if cycleNum >= 2
            priorCyclesMove := (cycleNum % 2 == 1) ? cycle0Net : 0.0

        cw1 = activeDir * len1
        cw2 = cw1 - activeDir * len2
        cw3 = cw2 + activeDir * len3
        cw4 = cw3 - activeDir * len4
        cw5 = cw4 + activeDir * len5

        cw2a = cw1 - activeDir * len1 * 0.382
        cw2b = cw2a + activeDir * (math.abs(cw1 - cw2a) * 0.5)
        cw2c = cw2

        cw4a = cw3 - activeDir * len3 * 0.236
        cw4b = cw4a + activeDir * (math.abs(cw3 - cw4a) * 0.5)
        cw4c = cw4

        float relTp = 0.0
        string lb = ""

        if cycleNum % 2 == 1
            // Corrective Cycle (A-B-C)
            if currIdx == 0
                relTp := cw1
                lb := "A"
            else if currIdx == 1
                // Retracements of A aren't typical numbered, skip sub-labels
                relTp := cw2a
                lb := ""
            else if currIdx == 2
                relTp := cw2b
                lb := ""
            else if currIdx == 3
                relTp := cw2c
                lb := "B"
            else if currIdx == 4
                relTp := cw3
                lb := "C"
            else if currIdx == 5
                relTp := cw4a
                lb := ""
            else if currIdx == 6
                relTp := cw4b
                lb := ""
            else if currIdx == 7
                relTp := cw4c
                lb := "X"
            else if currIdx == 8
                relTp := cw5
                lb := "Y"
        else
            // Motive Cycle (1-2-3-4-5)
            if currIdx == 0
                relTp := cw1
                lb := "(1)"
            else if currIdx == 1
                relTp := cw2a
                lb := "A"
            else if currIdx == 2
                relTp := cw2b
                lb := "B"
            else if currIdx == 3
                relTp := cw2c
                lb := "(2)"
            else if currIdx == 4
                relTp := cw3
                lb := "(3)"
            else if currIdx == 5
                relTp := cw4a
                lb := "A"
            else if currIdx == 6
                relTp := cw4b
                lb := "B"
            else if currIdx == 7
                relTp := cw4c
                lb := "(4)"
            else if currIdx == 8
                relTp := cw5
                lb := "(5)"

        tp = originP + priorCyclesMove + relTp
        nt = simT + barSp

        string finalLb = lb
        if lb != ""
            float pctChg = currentPrice != 0 ? ((tp - currentPrice) / currentPrice) * 100.0 : 0.0
            string pctStr = (pctChg >= 0 ? "+" : "") + str.tostring(math.round(pctChg * 10) / 10, "#.#") + "%"
            finalLb := lb + " (" + pctStr + ")"

        // Check if we are drawing the very first forward-looking line from current price
        if isFirstForwardLine
            // Always tie the first predictive line from live currentPrice
            line.new(startT, currentPrice, nt, tp, color=col, style=style, width=2)
            if lb != ""
                label.new(nt, tp, cycleNum > 0 ? finalLb : finalLb, color=col, style=label.style_label_left, textcolor=color.white, size=size.small)
            isFirstForwardLine := false
        else
            if lb != ""
                line.new(simT, simP, nt, tp, color=col, style=style, width=2)
                label.new(nt, tp, cycleNum > 0 ? finalLb : finalLb, color=col, style=label.style_label_left, textcolor=color.white, size=size.small)
            else
                // Just draw the line to the invisible intermediate point, but don't label it to reduce chart clutter for corrections
                line.new(simT, simP, nt, tp, color=col, style=style, width=2)

        simP := tp
        simT := nt

// â”€â”€â”€ Structural Wave Detection â”€â”€â”€
// Finds the 5 major EW turning points within RECENT pivots only.
// Only searches the last 20 pivots to stay anchored to current price action.
// Returns [originIdx, w1Idx, w2Idx, w3Idx, w4Idx]
findStructuralWaves(isBull) =>
    sz = array.size(px)
    recentStart = math.max(0, sz - 20)  // Only search last 20 pivots
    int oIdx = recentStart
    int w1I = recentStart
    int w2I = recentStart
    int w3I = recentStart
    int w4I = recentStart
    if sz >= 3
        if isBull
            // Step 1: Find highest pivot in recent window = W3
            w3I := recentStart
            maxVal = array.get(px, recentStart)
            for i = recentStart to sz - 1
                v = array.get(px, i)
                if v > maxVal
                    maxVal := v
                    w3I := i
            // Step 2: Lowest between recentStart and W3 = origin
            oIdx := recentStart
            minVal = array.get(px, recentStart)
            for i = recentStart to w3I
                v = array.get(px, i)
                if v < minVal
                    minVal := v
                    oIdx := i
            // Step 3: First local max between origin and W3 = W1
            w1I := math.min(oIdx + 1, w3I)
            if w3I > oIdx + 2
                localMax = array.get(px, oIdx + 1)
                for i = oIdx + 1 to w3I - 1
                    v = array.get(px, i)
                    if v > localMax
                        localMax := v
                        w1I := i
                    if v < localMax * 0.9
                        break
            // Step 4: Lowest between W1 and W3 = W2
            w2I := w1I
            if w3I > w1I + 1
                localMin = array.get(px, w1I)
                for i = w1I to w3I - 1
                    v = array.get(px, i)
                    if v < localMin
                        localMin := v
                        w2I := i
            // Step 5: Lowest after W3 = W4
            w4I := w3I
            if w3I < sz - 1
                localMin2 = array.get(px, w3I + 1)
                w4I := w3I + 1
                for i = w3I + 1 to sz - 1
                    v = array.get(px, i)
                    if v < localMin2
                        localMin2 := v
                        w4I := i
        else
            // Bear: highest in recent window = origin, then lowest after = W3
            oIdx := recentStart
            maxVal2 = array.get(px, recentStart)
            for i = recentStart to sz - 1
                v = array.get(px, i)
                if v > maxVal2
                    maxVal2 := v
                    oIdx := i
            // Lowest between origin and end of recent window = W3
            w3I := oIdx
            minVal2 = array.get(px, oIdx)
            for i = oIdx to sz - 1
                v = array.get(px, i)
                if v < minVal2
                    minVal2 := v
                    w3I := i
            // W1 = first significant trough between origin and W3
            w1I := math.min(oIdx + 1, w3I)
            if w3I > oIdx + 2
                localMin3 = array.get(px, oIdx + 1)
                for i = oIdx + 1 to w3I - 1
                    v = array.get(px, i)
                    if v < localMin3
                        localMin3 := v
                        w1I := i
                    if v > localMin3 * 1.1
                        break
            // W2 = highest bounce between W1 and W3
            w2I := w1I
            if w3I > w1I + 1
                localMax3 = array.get(px, w1I)
                for i = w1I to w3I - 1
                    v = array.get(px, i)
                    if v > localMax3
                        localMax3 := v
                        w2I := i
            // W4 = highest bounce after W3
            w4I := w3I
            if w3I < sz - 1
                localMax4 = array.get(px, w3I + 1)
                w4I := w3I + 1
                for i = w3I + 1 to sz - 1
                    v = array.get(px, i)
                    if v > localMax4
                        localMax4 := v
                        w4I := i
    [oIdx, w1I, w2I, w3I, w4I]

// â”€â”€â”€ Fibonacci Ratio Validation â”€â”€â”€
// Returns a confidence score 0-100 for how well the pivot sequence matches EW rules
ewConfidence(sz) =>
    float score = 0.0
    if sz >= 5
        // Get last 5 legs
        float l1 = math.abs(array.get(px, sz-5) - array.get(px, sz-4)) // Wave 1
        float l2 = math.abs(array.get(px, sz-4) - array.get(px, sz-3)) // Wave 2
        float l3 = math.abs(array.get(px, sz-3) - array.get(px, sz-2)) // Wave 3
        float l4 = math.abs(array.get(px, sz-2) - array.get(px, sz-1)) // Wave 4
        // Rule 1: Wave 3 is never the shortest (max +30 pts)
        if l3 >= l1 and l3 >= l4
            score += 30.0
        // Rule 2: Wave 2 retraces 38.2%-78.6% of Wave 1 (max +25 pts)
        if l1 > 0
            r2 = l2 / l1
            if r2 >= 0.382 and r2 <= 0.786
                score += 25.0
            else if r2 >= 0.236 and r2 <= 0.886
                score += 12.0
        // Rule 3: Wave 3 extends 1.272x-2.618x Wave 1 (max +25 pts)
        if l1 > 0
            ext3 = l3 / l1
            if ext3 >= 1.272 and ext3 <= 2.618
                score += 25.0
            else if ext3 >= 1.0 and ext3 <= 3.0
                score += 12.0
        // Rule 4: Wave 4 retraces 23.6%-50% of Wave 3 (max +20 pts)
        if l3 > 0
            r4 = l4 / l3
            if r4 >= 0.236 and r4 <= 0.500
                score += 20.0
            else if r4 >= 0.146 and r4 <= 0.618
                score += 10.0
    score
